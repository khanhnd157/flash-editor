⚡ Flash Editor — Custom Extensions

Extending the editor with a custom "Callout" block node and an "Emoji" inline node.

📄 Callout Extension (NodeExtension)

// Create a custom block extension
import { NodeExtension } from '@flash/core';

const Callout = NodeExtension.create({
  name: 'callout',

  addOptions: () => ({
    types: ['info', 'warning', 'error', 'success'],
  }),

  nodeSpec: () => ({
    group: 'block',
    content: 'block+',
    attrs: { type: { default: 'info' } },
    toDOM(node) {
      const icons = { info: 'ℹ️', warning: '⚠️', error: '❌', success: '✅' };
      return ['div', { class: 'callout', 'data-type': node.attrs.type },
        ['span', { class: 'callout-icon' }, icons[node.attrs.type]],
        ['div', { class: 'callout-content' }, 0],
      ];
    },
    parseDOM: [{ tag: 'div.callout', getAttrs(dom) {
      return { type: dom.getAttribute('data-type') || 'info' };
    }}],
  }),

  addCommands: () => ({
    insertCallout: (attrs) => (state, dispatch) => {
      // Insert a callout with a paragraph inside
      const calloutType = state.schema.nodes.callout;
      const paragraph = state.schema.nodes.paragraph.create();
      const node = calloutType.create(attrs, paragraph);
      const tr = state.tr.replaceSelectionWith(node);
      dispatch?.(tr);
      return true;
    },
    setCalloutType: (type) => (state, dispatch) => {
      // Change callout variant at cursor
      const { $from } = state.selection;
      // Walk up to find callout node
      for (let d = $from.depth; d > 0; d--) {
        if ($from.node(d).type.name === 'callout') {
          const pos = $from.before(d);
          dispatch?.(state.tr.setNodeMarkup(pos, undefined, { type }));
          return true;
        }
      }
      return false;
    },
  }),

  addKeyboardShortcuts: () => ({
    'Mod-Shift-c': (state, dispatch) => {
      // Quick insert info callout
      const callout = state.schema.nodes.callout;
      const para = state.schema.nodes.paragraph.create();
      dispatch?.(state.tr.replaceSelectionWith(callout.create({ type: 'info' }, para)));
      return true;
    },
  }),
});

📄 Emoji Extension (Inline NodeExtension)

const Emoji = NodeExtension.create({
  name: 'emoji',

  nodeSpec: () => ({
    group: 'inline',
    inline: true,
    atom: true,          // Treated as a single unit, not editable inside
    attrs: { code: {} },   // e.g. "rocket", "fire"
    toDOM(node) {
      const map = { rocket: '🚀', fire: '🔥', heart: '❤️', star: '⭐', check: '✅' };
      return ['span', { class: 'emoji-inline', 'data-emoji': node.attrs.code },
        map[node.attrs.code] || '❓'];
    },
    parseDOM: [{ tag: 'span.emoji-inline', getAttrs(dom) {
      return { code: dom.getAttribute('data-emoji') };
    }}],
  }),

  addCommands: () => ({
    insertEmoji: (code) => (state, dispatch) => {
      const emojiNode = state.schema.nodes.emoji.create({ code });
      dispatch?.(state.tr.replaceSelectionWith(emojiNode));
      return true;
    },
  }),
});
